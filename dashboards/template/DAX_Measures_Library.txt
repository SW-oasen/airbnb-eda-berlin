// ===================================================================
// PRICING MEASURES
// ===================================================================

Average Price = 
AVERAGE(listings_cleaned[price])

Median Price = 
MEDIAN(listings_cleaned[price])

Price Range = 
MAX(listings_cleaned[price]) - MIN(listings_cleaned[price])

Total Revenue Potential = 
SUM(listings_cleaned[price]) * 30

Price per Guest = 
DIVIDE(
    AVERAGE(listings_cleaned[price]), 
    AVERAGE(listings_cleaned[accommodates]), 
    0
)

Price Category = 
SWITCH(
    TRUE(),
    AVERAGE(listings_cleaned[price]) <= 50, "Budget (≤€50)",
    AVERAGE(listings_cleaned[price]) <= 100, "Mid-range (€51-€100)",
    AVERAGE(listings_cleaned[price]) <= 200, "Premium (€101-€200)",
    "Luxury (€200+)"
)

Price Percentile 75 = 
PERCENTILE.INC(listings_cleaned[price], 0.75)

Price Percentile 25 = 
PERCENTILE.INC(listings_cleaned[price], 0.25)

Above Average Price Count = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[price] > [Average Price]
)

Revenue per Square Meter = 
DIVIDE(
    [Average Price],
    AVERAGE(listings_cleaned[accommodates]),
    0
) * 30

Price Competitiveness = 
DIVIDE(
    AVERAGE(listings_cleaned[price]),
    CALCULATE([Average Price], ALL(listings_cleaned)),
    1
)

// ===================================================================
// AVAILABILITY MEASURES
// ===================================================================

Average Availability = 
AVERAGE(listings_cleaned[availability_365])

Median Availability = 
MEDIAN(listings_cleaned[availability_365])

High Availability Listings = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[availability_365] > 300
)

Low Availability Listings = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[availability_365] < 100
)

Occupancy Rate = 
DIVIDE(
    365 - AVERAGE(listings_cleaned[availability_365]),
    365,
    0
) * 100

Booking Efficiency = 
DIVIDE(
    365 - AVERAGE(listings_cleaned[availability_365]),
    365
) * 100

Estimated Annual Revenue = 
AVERAGE(listings_cleaned[price]) * (365 - AVERAGE(listings_cleaned[availability_365]))

Availability Category = 
SWITCH(
    TRUE(),
    AVERAGE(listings_cleaned[availability_365]) >= 300, "High Availability (300+ days)",
    AVERAGE(listings_cleaned[availability_365]) >= 180, "Medium Availability (180-299 days)",
    AVERAGE(listings_cleaned[availability_365]) >= 90, "Low Availability (90-179 days)",
    "Very Low Availability (<90 days)"
)

// ===================================================================
// HOST PERFORMANCE MEASURES
// ===================================================================

Total Hosts = 
DISTINCTCOUNT(listings_cleaned[host_id])

Average Reviews per Listing = 
DIVIDE(
    SUM(listings_cleaned[number_of_reviews]),
    COUNT(listings_cleaned[id]),
    0
)

Superhosts Count = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[host_is_superhost] = "t"
)

Superhosts Percentage = 
DIVIDE(
    [Superhosts Count],
    COUNT(listings_cleaned[id]),
    0
) * 100

Multi-listing Hosts = 
CALCULATE(
    DISTINCTCOUNT(listings_cleaned[host_id]),
    listings_cleaned[calculated_host_listings_count] > 1
)

Average Host Listings = 
AVERAGE(listings_cleaned[calculated_host_listings_count])

Superhost Premium = 
CALCULATE([Average Price], listings_cleaned[host_is_superhost] = "t") - 
CALCULATE([Average Price], listings_cleaned[host_is_superhost] = "f")

Reviews per Month = 
DIVIDE(
    AVERAGE(listings_cleaned[reviews_per_month]),
    1,
    0
)

Host Response Rate = 
AVERAGE(VALUE(SUBSTITUTE(listings_cleaned[host_response_rate], "%", "")))

Identity Verified Hosts = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[host_identity_verified] = "t"
)

Host Performance Score = 
(
    DIVIDE([Host Response Rate], 100, 0) * 0.3 +
    DIVIDE([Reviews per Month], 5, 0) * 0.4 +
    IF(AVERAGE(listings_cleaned[host_is_superhost]) = "t", 1, 0) * 0.3
) * 100

// ===================================================================
// REVIEW AND RATING MEASURES
// ===================================================================

Average Rating = 
AVERAGE(listings_cleaned[review_scores_rating])

High Rated Listings = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[review_scores_rating] >= 90
)

Review Count = 
SUM(listings_cleaned[number_of_reviews])

Average Reviews Score = 
AVERAGE(listings_cleaned[review_scores_rating])

Cleanliness Score = 
AVERAGE(listings_cleaned[review_scores_cleanliness])

Communication Score = 
AVERAGE(listings_cleaned[review_scores_communication])

Location Score = 
AVERAGE(listings_cleaned[review_scores_location])

Value Score = 
AVERAGE(listings_cleaned[review_scores_value])

Recent Reviews = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[last_review] >= DATE(2024, 1, 1)
)

// ===================================================================
// GEOGRAPHIC MEASURES
// ===================================================================

District Category = 
SWITCH(
    SELECTEDVALUE(listings_cleaned[neighbourhood_group]),
    "Mitte", "Central Berlin",
    "Friedrichshain-Kreuzberg", "Central Berlin",
    "Pankow", "North Berlin",
    "Charlottenburg-Wilmersdorf", "West Berlin",
    "Neukölln", "South Berlin",
    "Tempelhof-Schöneberg", "South Berlin",
    "Steglitz-Zehlendorf", "Southwest Berlin",
    "Spandau", "West Berlin",
    "Reinickendorf", "North Berlin",
    "Marzahn-Hellersdorf", "East Berlin",
    "Lichtenberg", "East Berlin",
    "Treptow-Köpenick", "Southeast Berlin",
    "Other"
)

Listings per District = 
COUNT(listings_cleaned[id])

District Market Share = 
DIVIDE(
    COUNT(listings_cleaned[id]),
    CALCULATE(COUNT(listings_cleaned[id]), ALL(listings_cleaned[neighbourhood_group])),
    0
) * 100

Central Location Premium = 
CALCULATE([Average Price], [District Category] = "Central Berlin") - [Average Price]

Competition Index = 
DIVIDE(
    COUNT(listings_cleaned[id]),
    DISTINCTCOUNT(listings_cleaned[neighbourhood_cleansed])
)

// ===================================================================
// PROPERTY TYPE MEASURES
// ===================================================================

Entire Home Count = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[room_type] = "Entire home/apt"
)

Private Room Count = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[room_type] = "Private room"
)

Shared Room Count = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[room_type] = "Shared room"
)

Hotel Room Count = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[room_type] = "Hotel room"
)

Room Type Distribution = 
DIVIDE(
    COUNT(listings_cleaned[id]),
    CALCULATE(COUNT(listings_cleaned[id]), ALL(listings_cleaned[room_type])),
    0
) * 100

Average Accommodates = 
AVERAGE(listings_cleaned[accommodates])

Large Properties = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[accommodates] >= 6
)

// ===================================================================
// AMENITIES AND FEATURES MEASURES
// ===================================================================

Instant Bookable = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[instant_bookable] = "t"
)

Instant Bookable Percentage = 
DIVIDE(
    [Instant Bookable],
    COUNT(listings_cleaned[id]),
    0
) * 100

Minimum Nights Average = 
AVERAGE(listings_cleaned[minimum_nights])

Maximum Nights Average = 
AVERAGE(listings_cleaned[maximum_nights])

Long Stay Properties = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[minimum_nights] >= 7
)

Short Stay Properties = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[minimum_nights] <= 3
)

// ===================================================================
// BUSINESS INTELLIGENCE MEASURES
// ===================================================================

Market Opportunity Score = 
(
    DIVIDE([Occupancy Rate], 80, 0) * 0.3 +
    DIVIDE([Average Rating], 95, 0) * 0.2 +
    DIVIDE([Reviews per Month], 3, 0) * 0.2 +
    DIVIDE([Average Price], 150, 0) * 0.3
) * 100

Profitability Index = 
DIVIDE(
    [Estimated Annual Revenue],
    50000,  -- Assuming €50k baseline
    0
)

Investment Attractiveness = 
SWITCH(
    TRUE(),
    [Market Opportunity Score] >= 80, "Excellent",
    [Market Opportunity Score] >= 60, "Good",
    [Market Opportunity Score] >= 40, "Fair",
    "Poor"
)

Price-to-Rating Ratio = 
DIVIDE(
    AVERAGE(listings_cleaned[price]),
    AVERAGE(listings_cleaned[review_scores_rating]),
    0
)

Demand Indicator = 
(
    (100 - AVERAGE(listings_cleaned[availability_365])) / 365 * 0.4 +
    AVERAGE(listings_cleaned[reviews_per_month]) / 5 * 0.3 +
    AVERAGE(listings_cleaned[number_of_reviews]) / 100 * 0.3
) * 100

// ===================================================================
// TIME-BASED MEASURES (if date fields available)
// ===================================================================

Days Since Last Review = 
DATEDIFF(
    MAX(listings_cleaned[last_review]),
    TODAY(),
    DAY
)

Active Listings = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[last_review] >= DATE(2024, 1, 1)
)

Recently Listed = 
CALCULATE(
    COUNT(listings_cleaned[id]),
    listings_cleaned[host_since] >= DATE(2023, 1, 1)
)

// ===================================================================
// CONDITIONAL FORMATTING MEASURES
// ===================================================================

Price Color = 
SWITCH(
    TRUE(),
    [Average Price] >= 150, "#FF6B6B",  -- Red for high prices
    [Average Price] >= 100, "#FFE66D",  -- Yellow for medium prices
    "#4ECDC4"  -- Teal for low prices
)

Occupancy Color = 
SWITCH(
    TRUE(),
    [Occupancy Rate] >= 70, "#51CF66",  -- Green for high occupancy
    [Occupancy Rate] >= 50, "#FFE066",  -- Yellow for medium occupancy
    "#FF6B6B"  -- Red for low occupancy
)

Rating Color = 
SWITCH(
    TRUE(),
    [Average Rating] >= 90, "#51CF66",  -- Green for excellent ratings
    [Average Rating] >= 80, "#FFE066",  -- Yellow for good ratings
    "#FF6B6B"  -- Red for poor ratings
)

// ===================================================================
// RANKING MEASURES
// ===================================================================

Price Rank = 
RANKX(
    ALL(listings_cleaned),
    [Average Price],
    ,
    DESC
)

Revenue Rank = 
RANKX(
    ALL(listings_cleaned),
    [Estimated Annual Revenue],
    ,
    DESC
)

Rating Rank = 
RANKX(
    ALL(listings_cleaned),
    [Average Rating],
    ,
    DESC
)

District Price Rank = 
RANKX(
    ALL(listings_cleaned[neighbourhood_group]),
    [Average Price],
    ,
    DESC
)

// ===================================================================
// WHAT-IF ANALYSIS MEASURES
// ===================================================================

Price Adjustment Impact = 
[Average Price] * (1 + 'Price Adjustment'[Price Adjustment Value])

Occupancy Impact = 
[Occupancy Rate] * (1 + 'Occupancy Adjustment'[Occupancy Adjustment Value])

Revenue Scenario = 
[Price Adjustment Impact] * ([Occupancy Impact] / 100) * 365
